#!/usr/bin/env bash
#==============================================================================
# river-desktop-setup.sh — River (Wayland) desktop stack for Arch Linux
#
# Run as your NORMAL user (with sudo rights) on a fresh or existing Arch
# installation:
#
#     bash river-desktop-setup.sh
#
# What it does:
#   - Full system upgrade
#   - Auto-detects environment (VirtualBox / VMware / QEMU-KVM / bare metal)
#     and GPU vendor (AMD / Intel / NVIDIA) and installs matching drivers
#   - Installs River + a complete Wayland desktop stack:
#       river, foot, fuzzel, waybar, mako, swaybg, swaylock, swayidle,
#       grim, slurp, wl-clipboard, cliphist, PipeWire, WirePlumber,
#       xdg-desktop-portal(-wlr/-gtk), polkit-gnome, NetworkManager applet
#   - Installs and enables greetd + tuigreet (Wayland-native login manager)
#   - Writes ready-to-use configs for river, waybar, fuzzel, mako, foot,
#     swaylock and the xdg portals
#
# Existing config files are backed up with a .bak-YYYYMMDD-HHMMSS suffix.
#==============================================================================

set -euo pipefail

c_green=$'\e[32m'; c_red=$'\e[31m'; c_yellow=$'\e[33m'; c_cyan=$'\e[36m'; c_reset=$'\e[0m'
info() { printf '%s==>%s %s\n' "$c_green" "$c_reset" "$*"; }
warn() { printf '%s!%s   %s\n' "$c_yellow" "$c_reset" "$*"; }
die()  { printf '%sERROR:%s %s\n' "$c_red" "$c_reset" "$*" >&2; exit 1; }

[[ $EUID -ne 0 ]] || die "Run this as your normal user, not root."
sudo -v || die "This user needs sudo rights."
command -v pacman >/dev/null || die "This script targets Arch Linux (pacman not found)."

BAK=".bak-$(date +%Y%m%d-%H%M%S)"

backup() {  # backup <path> if it exists
    local p=$1
    if [[ -e $p ]]; then
        warn "Backing up existing $p -> ${p}${BAK}"
        mv "$p" "${p}${BAK}"
    fi
}

# --- 1. System upgrade ---------------------------------------------------------
info "Performing full system upgrade..."
sudo pacman -Syu --needed --noconfirm base-devel git

# --- 2. Detect environment & GPU -----------------------------------------------
VIRT=$(systemd-detect-virt 2>/dev/null || echo none)
info "Virtualization: $VIRT"

GPU="generic"
if   lspci | grep -qi nvidia;                                    then GPU=nvidia;
elif lspci | grep -qiE 'amd|ati';                                then GPU=amd;
elif lspci | grep -qiE 'intel.*(vga|3d|display|graphics)';       then GPU=intel; fi
info "Detected GPU class: $GPU"

GPU_PKGS=(mesa)
GPUENV=""   # extra environment written into the river session wrapper

case $VIRT in
    oracle)
        info "VirtualBox detected — installing guest utilities."
        GPU_PKGS+=(virtualbox-guest-utils)
        ;;
    vmware)
        info "VMware detected — installing open-vm-tools."
        GPU_PKGS+=(open-vm-tools)
        ;;
    kvm|qemu)
        info "QEMU/KVM detected — installing SPICE guest agent."
        GPU_PKGS+=(spice-vdagent qemu-guest-agent)
        ;;
    none)
        case $GPU in
            amd)
                info "AMD GPU — installing Vulkan and video-acceleration drivers."
                GPU_PKGS+=(vulkan-radeon libva-mesa-driver mesa-vdpau)
                ;;
            intel)
                info "Intel GPU — installing Vulkan and video-acceleration drivers."
                GPU_PKGS+=(vulkan-intel intel-media-driver libva-intel-driver)
                ;;
            nvidia)
                info "NVIDIA GPU — installing proprietary driver (nvidia-open)."
                GPU_PKGS+=(nvidia-open nvidia-utils libva-nvidia-driver)
                GPUENV='export LIBVA_DRIVER_NAME=nvidia
export __GLX_VENDOR_LIBRARY_NAME=nvidia
export GBM_BACKEND=nvidia-drm
export WLR_NO_HARDWARE_CURSORS=1'
                ;;
            *)
                warn "GPU vendor not recognized — installing generic Mesa stack only."
                ;;
        esac
        ;;
    *)
        warn "Unknown hypervisor '$VIRT' — installing generic Mesa stack only."
        ;;
esac

# --- 3. Package installation -----------------------------------------------------
# river vs river-classic: use whichever the repos currently provide.
if pacman -Si river &>/dev/null; then RIVER_PKG=river;
elif pacman -Si river-classic &>/dev/null; then RIVER_PKG=river-classic;
else die "Neither 'river' nor 'river-classic' found in the repos. Is the system up to date?"; fi
info "River package: $RIVER_PKG"

CORE_PKGS=(
    "$RIVER_PKG"            # the wlroots-based Wayland compositor
    foot                    # Wayland-native terminal
    fuzzel                  # Wayland-native app launcher / dmenu
    waybar                  # status bar (has river/* modules)
    mako                    # notification daemon
    swaybg                  # wallpaper / background
    swaylock                # screen locker
    swayidle                # idle management
    grim slurp              # screenshots
    wl-clipboard cliphist   # clipboard + history
    xdg-user-dirs xdg-utils
    xdg-desktop-portal xdg-desktop-portal-wlr xdg-desktop-portal-gtk
    polkit-gnome            # authentication agent
    network-manager-applet  # nm-applet tray icon
    brightnessctl           # backlight control
    playerctl               # media keys
    libnotify               # notify-send
)
AUDIO_PKGS=(pipewire pipewire-alsa pipewire-pulse pipewire-jack wireplumber pavucontrol)
FONT_PKGS=(ttf-dejavu noto-fonts noto-fonts-emoji ttf-jetbrains-mono-nerd otf-font-awesome)
DM_PKGS=(greetd greetd-tuigreet)

read -rp "Install extras (firefox, thunar, blueman, btop)? [Y/n]: " EXTRAS; EXTRAS=${EXTRAS:-Y}
EXTRA_PKGS=()
if [[ ${EXTRAS^} == Y* ]]; then
    EXTRA_PKGS=(firefox thunar gvfs blueman bluez bluez-utils btop)
fi

info "Installing packages (this will take a while)..."
sudo pacman -S --needed --noconfirm \
    "${GPU_PKGS[@]}" "${CORE_PKGS[@]}" "${AUDIO_PKGS[@]}" \
    "${FONT_PKGS[@]}" "${DM_PKGS[@]}" "${EXTRA_PKGS[@]}"

# --- 4. Services -----------------------------------------------------------------
info "Enabling system services..."
sudo systemctl enable NetworkManager.service 2>/dev/null || true
if [[ $VIRT == oracle ]]; then sudo systemctl enable vboxservice.service; fi
if [[ $VIRT == vmware ]]; then sudo systemctl enable vmtoolsd.service; fi
if [[ $VIRT == kvm || $VIRT == qemu ]]; then sudo systemctl enable qemu-guest-agent.service 2>/dev/null || true; fi
if printf '%s\n' "${EXTRA_PKGS[@]}" | grep -qx bluez; then
    sudo systemctl enable bluetooth.service
fi

# PipeWire runs as a user service — enable it globally for all users.
info "Enabling PipeWire user services (globally)..."
sudo systemctl --global enable pipewire.service pipewire-pulse.service wireplumber.service

# --- 5. NVIDIA-specific system tweaks --------------------------------------------
if [[ $GPU == nvidia && $VIRT == none ]]; then
    info "Applying NVIDIA Wayland tweaks (kernel mode setting, early KMS)..."
    if ! grep -q nvidia_drm /etc/mkinitcpio.conf; then
        sudo sed -i 's/^MODULES=(/MODULES=(nvidia nvidia_modeset nvidia_uvm nvidia_drm /' /etc/mkinitcpio.conf
        sudo mkinitcpio -P
    fi
    for entry in /boot/loader/entries/*.conf; do
        if ! grep -q nvidia_drm.modeset "$entry"; then
            sudo sed -i '/^options / s/$/ nvidia_drm.modeset=1 nvidia_drm.fbdev=1/' "$entry"
        fi
    done
fi

#==============================================================================
# 6. User configuration files
#==============================================================================
info "Writing user configuration files..."
mkdir -p "$HOME/.config/river" "$HOME/.config/waybar" "$HOME/.config/fuzzel" \
         "$HOME/.config/mako" "$HOME/.config/foot" "$HOME/.config/swaylock" \
         "$HOME/.config/xdg-desktop-portal"
xdg-user-dirs-update
mkdir -p "$HOME/Pictures/Screenshots"

# ---- river/init ---------------------------------------------------------------
backup "$HOME/.config/river/init"
cat > "$HOME/.config/river/init" <<'EOF'
#!/bin/sh
#==============================================================================
# River configuration — ~/.config/river/init
# Super (Windows key) is the main modifier.
# NOTE: re-running this file while river is running re-applies settings but
# also spawns autostart programs again — log out/in for a clean restart.
#==============================================================================

# ---- Autostart ----------------------------------------------------------------
riverctl spawn "dbus-update-activation-environment --systemd WAYLAND_DISPLAY XDG_CURRENT_DESKTOP"
riverctl spawn "swaybg -c '#1e1e2e'"          # solid color; replace with: swaybg -i ~/Pictures/wallpaper.jpg -m fill
riverctl spawn "waybar"
riverctl spawn "mako"
riverctl spawn "swayidle -w timeout 600 'swaylock -f' before-sleep 'swaylock -f'"
riverctl spawn "/usr/lib/polkit-gnome/polkit-gnome-authentication-agent-1"
riverctl spawn "wl-paste --type text  --watch cliphist store"
riverctl spawn "wl-paste --type image --watch cliphist store"
riverctl spawn "nm-applet --indicator"
riverctl spawn "command -v blueman-applet >/dev/null && exec blueman-applet"

# ---- Layout generator (rivertile) ---------------------------------------------
riverctl default-layout rivertile
riverctl spawn "rivertile -view-padding 4 -outer-padding 4 -main-ratio 0.55"

# ---- Application launchers ------------------------------------------------------
riverctl map normal Super Return spawn foot
riverctl map normal Super D spawn fuzzel
riverctl map normal Super V spawn 'cliphist list | fuzzel --dmenu | cliphist decode | wl-copy'

# ---- Window management ----------------------------------------------------------
riverctl map normal Super Q close
riverctl map normal Super+Shift E exit            # quit river (back to login)
riverctl map normal Super J focus-view next
riverctl map normal Super K focus-view previous
riverctl map normal Super+Shift J swap next
riverctl map normal Super+Shift K swap previous
riverctl map normal Super+Shift Return zoom       # bump focused view to main
riverctl map normal Super Space toggle-float
riverctl map normal Super F toggle-fullscreen

# ---- rivertile layout control ---------------------------------------------------
riverctl map normal Super H send-layout-cmd rivertile "main-ratio -0.05"
riverctl map normal Super L send-layout-cmd rivertile "main-ratio +0.05"
riverctl map normal Super+Shift H send-layout-cmd rivertile "main-count +1"
riverctl map normal Super+Shift L send-layout-cmd rivertile "main-count -1"

# ---- Multi-monitor ---------------------------------------------------------------
riverctl map normal Super Tab focus-output next
riverctl map normal Super+Shift Tab send-to-output next

# ---- Tags (workspaces) ------------------------------------------------------------
for i in $(seq 1 9); do
    tags=$((1 << (i - 1)))
    riverctl map normal Super "$i" set-focused-tags "$tags"                 # go to tag
    riverctl map normal Super+Shift "$i" set-view-tags "$tags"              # move window to tag
    riverctl map normal Super+Control "$i" toggle-focused-tags "$tags"      # show additionally
    riverctl map normal Super+Shift+Control "$i" toggle-view-tags "$tags"   # pin window to tag
done
all_tags=$(((1 << 32) - 1))
riverctl map normal Super 0 set-focused-tags "$all_tags"                    # show all
riverctl map normal Super+Shift 0 set-view-tags "$all_tags"                 # pin to all

# ---- Screenshots -------------------------------------------------------------------
riverctl map normal None Print spawn 'grim - | wl-copy && notify-send "Screenshot" "Copied to clipboard"'
riverctl map normal Shift Print spawn 'grim -g "$(slurp -d)" - | wl-copy && notify-send "Screenshot" "Selection copied"'
riverctl map normal Control Print spawn 'grim "$HOME/Pictures/Screenshots/$(date +%F_%H-%M-%S).png" && notify-send "Screenshot" "Saved to ~/Pictures/Screenshots"'

# ---- Lock, volume, brightness, media ------------------------------------------------
riverctl map normal Control+Alt L spawn "swaylock -f"
riverctl map normal None XF86AudioRaiseVolume spawn 'wpctl set-volume -l 1 @DEFAULT_AUDIO_SINK@ 5%+'
riverctl map normal None XF86AudioLowerVolume spawn 'wpctl set-volume @DEFAULT_AUDIO_SINK@ 5%-'
riverctl map normal None XF86AudioMute spawn 'wpctl set-mute @DEFAULT_AUDIO_SINK@ toggle'
riverctl map normal None XF86AudioMicMute spawn 'wpctl set-mute @DEFAULT_AUDIO_SOURCE@ toggle'
riverctl map normal None XF86MonBrightnessUp spawn 'brightnessctl set 10%+'
riverctl map normal None XF86MonBrightnessDown spawn 'brightnessctl set 10%-'
riverctl map normal None XF86AudioPlay spawn 'playerctl play-pause'
riverctl map normal None XF86AudioNext spawn 'playerctl next'
riverctl map normal None XF86AudioPrev spawn 'playerctl previous'

# ---- Mouse ---------------------------------------------------------------------------
riverctl map-pointer normal Super BTN_LEFT move-view
riverctl map-pointer normal Super BTN_RIGHT resize-view
riverctl map-pointer normal Super BTN_MIDDLE toggle-float

# ---- Input devices ---------------------------------------------------------------------
riverctl keyboard-layout -options caps:escape us
riverctl set-repeat 40 300
riverctl list-inputs | grep -i touchpad | while IFS= read -r pad; do
    riverctl input "$pad" tap enabled
    riverctl input "$pad" tap-button-map lrm
    riverctl input "$pad" natural-scroll enabled
    riverctl input "$pad" disable-while-typing enabled
done

# ---- Behavior & appearance ---------------------------------------------------------------
riverctl focus-follows-cursor normal
riverctl set-cursor-warp on-output-change
riverctl attach-mode bottom
riverctl border-color-focused 0x89b4fa
riverctl border-color-unfocused 0x313244
riverctl border-color-urgent 0xf38ba8
riverctl border-width 2

# ---- Window rules ---------------------------------------------------------------------------
riverctl rule-add -app-id '*' ssd                                   # server-side decorations everywhere
riverctl rule-add -app-id 'org.pulseaudio.pavucontrol' float
riverctl rule-add -app-id 'org.gnome.Calculator' float
riverctl rule-add -title 'Picture in Picture' float
EOF
chmod +x "$HOME/.config/river/init"

# ---- waybar ------------------------------------------------------------------------
backup "$HOME/.config/waybar/config.jsonc"
cat > "$HOME/.config/waybar/config.jsonc" <<'EOF'
{
    "layer": "top",
    "position": "top",
    "height": 30,
    "spacing": 4,
    "modules-left": ["river/tags", "river/mode", "river/window"],
    "modules-center": ["clock"],
    "modules-right": ["tray", "cpu", "memory", "pulseaudio", "network", "battery"],

    "river/tags":  { "num-tags": 9 },
    "river/mode":  { "format": "<span style=\"italic\">{}</span>" },
    "river/window": { "format": "{}", "max-length": 60 },

    "clock": {
        "format": "{:%a %d %b  %H:%M}",
        "tooltip-format": "<big>{:%Y %B}</big>\n<tt><small>{calendar}</small></tt>"
    },
    "cpu":    { "format": "󰻠 {usage}%", "tooltip": false },
    "memory": { "format": "󰍛 {}%" },
    "pulseaudio": {
        "format": "{icon} {volume}%",
        "format-muted": "󰝟 muted",
        "format-icons": { "default": ["󰕿", "󰖀", "󰕾"] },
        "scroll-step": 5,
        "on-click": "pavucontrol"
    },
    "network": {
        "format-wifi": "󰤨 {essid} ({signalStrength}%)",
        "format-ethernet": "󰈀 wired",
        "format-disconnected": "󰤭 offline",
        "on-click": "nm-connection-editor"
    },
    "battery": {
        "states": { "warning": 30, "critical": 15 },
        "format": "{icon} {capacity}%",
        "format-icons": ["󰂎", "󰁺", "󰁻", "󰁼", "󰁽", "󰁾", "󰁿", "󰂀", "󰂁", "󰂂", "󰁹"]
    },
    "tray": { "spacing": 8 }
}
EOF

backup "$HOME/.config/waybar/style.css"
cat > "$HOME/.config/waybar/style.css" <<'EOF'
* {
    font-family: "JetBrainsMono Nerd Font", "Font Awesome 6 Free", sans-serif;
    font-size: 13px;
    min-height: 0;
}

    background: rgba(30, 30, 46, 0.92);
    color: #cdd6f4;
    border-bottom: 2px solid #313244;
}
#tags button {
    padding: 0 8px;
    color: #6c7086;
}
#tags button.focused {
    color: #1e1e2e;
    background: #89b4fa;
    border-radius: 4px;
}
#tags button.urgent { color: #f38ba8; }
#window { padding: 0 10px; color: #a6adc8; }
#clock { color: #f9e2af; }
#cpu, #memory, #pulseaudio, #network, #battery, #tray { padding: 0 10px; }
#battery.warning { color: #f9e2af; }
#battery.critical { color: #f38ba8; }
EOF

# ---- fuzzel ------------------------------------------------------------------------
backup "$HOME/.config/fuzzel/fuzzel.ini"
cat > "$HOME/.config/fuzzel/fuzzel.ini" <<'EOF'

# ---- mako --------------------------------------------------------------------------
backup "$HOME/.config/mako/config"
cat > "$HOME/.config/mako/config" <<'EOF'

backup "$HOME/.config/swaylock/config"

# ---- xdg-desktop-portal --------------------------------------------------------------
backup "$HOME/.config/xdg-desktop-portal/portals.conf"
cat > "$HOME/.config/xdg-desktop-portal/portals.conf" <<'EOF'
[preferred]
default=gtk
org.freedesktop.impl.portal.ScreenCast=wlr
org.freedesktop.impl.portal.Screenshot=wlr
EOF

#==============================================================================
# 7. greetd + tuigreet (login manager) and the river session wrapper
#==============================================================================
info "Setting up greetd + tuigreet..."

# Session wrapper: sets the Wayland environment, then starts river.
cat <<EOF | sudo tee /usr/local/bin/river-session >/dev/null
#!/bin/sh
# Session wrapper started by greetd/tuigreet. Sets the environment for the
# whole Wayland session, then replaces itself with river.
export XDG_SESSION_TYPE=wayland
export XDG_SESSION_DESKTOP=river
export XDG_CURRENT_DESKTOP=river
export MOZ_ENABLE_WAYLAND=1
export QT_QPA_PLATFORM=wayland
export QT_WAYLAND_DISABLE_WINDOWDECORATION=1
export SDL_VIDEODRIVER=wayland
export CLUTTER_BACKEND=wayland
export _JAVA_AWT_WM_NONREPARENTING=1
$GPUENV
exec river
EOF
sudo chmod +x /usr/local/bin/river-session

# greetd needs an unprivileged 'greeter' user (the Arch package normally
# creates it — create it if missing).
if ! id greeter &>/dev/null; then
    sudo useradd -r -s /usr/bin/nologin -M greeter
fi

cat <<'EOF' | sudo tee /etc/greetd/config.toml >/dev/null
[terminal]
# The virtual terminal where the greeter should run.
vt = 1

[default_session]
# tuigreet: a console greeter UI. --remember remembers your last session,
# --time shows a clock, --user-menu lets you pick the user.
command = "tuigreet --time --remember --user-menu --cmd /usr/local/bin/river-session"
user = "greeter"
EOF

# Avoid a conflict between greetd on tty1 and the default getty.
sudo systemctl disable getty@tty1.service 2>/dev/null || true
# Disable other display managers if one was enabled before.
for dm in sddm gdm lightdm lxdm ly; do
    if systemctl is-enabled "$dm.service" &>/dev/null; then
        warn "Disabling previously enabled display manager: $dm"
        sudo systemctl disable "$dm.service"
    fi
done
sudo systemctl enable greetd.service

#==============================================================================
# 8. Done
#==============================================================================
echo
info "All done! Reboot to land in the tuigreet login screen:   sudo reboot"
cat <<'EOF'

  ------------------------------------------------------------------
  Essential key bindings (Super = Windows key)
  ------------------------------------------------------------------
    Super + Return .......... terminal (foot)
    Super + D ............... application launcher (fuzzel)
    Super + V ............... clipboard history picker
    Super + Q ............... close window
    Super + Shift + E ....... log out (exit river)
    Ctrl + Alt + L .......... lock screen (swaylock)
    Super + J / K ........... focus next / previous window
    Super + Shift + J / K ... swap window
    Super + H / L ........... shrink / grow main area
    Super + 1-9 ............. switch to tag (workspace)
    Super + Shift + 1-9 ..... move window to tag
    Super + F ............... fullscreen
    Super + Space ........... floating on/off
    Print ................... screenshot -> clipboard
    Shift + Print ........... area screenshot -> clipboard
    Ctrl + Print ............ screenshot -> ~/Pictures/Screenshots
  ------------------------------------------------------------------
  Config files live in ~/.config/{river,waybar,fuzzel,mako,foot,swaylock}
  Set a wallpaper by editing ~/.config/river/init (swaybg line).
  ------------------------------------------------------------------

EOF
